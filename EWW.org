(defvar workspace-object-array `[
   {"symbol": "", "number": 1},
   {"symbol": "", "number": 2},
   {"symbol": "", "number": 3},
   {"symbol": "󰈹", "number": 4},
   {"symbol": "", "number": 5},
   {"symbol": "", "number": 6}
]`)

(defwidget workspaces []
  (box
   :orientation "horizontal"
   :halign "start"
   (for workspace in workspace-object-array
        (button
         :onclick `swaymsg workspace ${workspace.number}`
         `${workspace.symbol}`))))
